    48| def wls_line(y, se, x, valid):
    49|     """Vectorized weighted least squares of y ~ 1 + x, weights 1/se^2.
    50| 
    51|     Exact maximum likelihood for Gaussian error with known se, which is what
    52|     fitModels.R optimizes numerically. Returns (intercept, slope, df).
    53|     """
    54|     w = np.where(valid, 1.0 / np.where(valid, se, 1.0) ** 2, 0.0)
    55|     yy = np.where(valid, y, 0.0)
    56|     xx = np.broadcast_to(x, y.shape)
    57|     S0 = w.sum(axis=1)
    58|     S1 = (w * xx).sum(axis=1)
    59|     S2 = (w * xx**2).sum(axis=1)
    60|     T0 = (w * yy).sum(axis=1)
    61|     T1 = (w * xx * yy).sum(axis=1)
    62|     det = S0 * S2 - S1**2
    63|     with np.errstate(divide="ignore", invalid="ignore"):
    64|         base = np.where(det != 0, (S2 * T0 - S1 * T1) / det, np.nan)
    65|         slope = np.where(det != 0, (S0 * T1 - S1 * T0) / det, np.nan)
    66|     return base, slope, valid.sum(axis=1) - 2
