  --- wls_line -- the weighted aggregation: scripts/lib/own_context.py lines 48-66 (verbatim from disk) ---
    48| def wls_line(y, se, x, valid):
    49|     """Vectorized weighted least squares of y ~ 1 + x, weights 1/se^2.
    50| 
    51|     Exact maximum likelihood for Gaussian error with known se, which is what
    52|     fitModels.R optimizes numerically. Returns (intercept, slope, df).
    53|     """
    54|     w = np.where(valid, 1.0 / np.where(valid, se, 1.0) ** 2, 0.0)
    55|     yy = np.where(valid, y, 0.0)
    56|     xx = np.broadcast_to(x, y.shape)
    57|     S0 = w.sum(axis=1)
    58|     S1 = (w * xx).sum(axis=1)
    59|     S2 = (w * xx**2).sum(axis=1)
    60|     T0 = (w * yy).sum(axis=1)
    61|     T1 = (w * xx * yy).sum(axis=1)
    62|     det = S0 * S2 - S1**2
    63|     with np.errstate(divide="ignore", invalid="ignore"):
    64|         base = np.where(det != 0, (S2 * T0 - S1 * T1) / det, np.nan)
    65|         slope = np.where(det != 0, (S0 * T1 - S1 * T0) / det, np.nan)
    66|     return base, slope, valid.sum(axis=1) - 2

   143| def fit_interaction(m_score, m_se, w_fitness, w_remediation, w_post, w_mean_score,
   144|                     a222v_fitness, a222v_remediation, concs=CONCS, correction=None):
   145|     """Fit e.b / e.r: the deviation of the A222V-background arm from the
   146|     multiplicative no-interaction expectation.
   147| 
   148|     expected(c) = single_mutant_term(c) * (b_A222V + c*r_A222V) [+ correction]
   149|     where single_mutant_term is the fitted line when w.post > 0.5, otherwise
   150|     the variant's mean WT-background score (per fitModels.R).
   151|     """
   152|     a222v_line = a222v_fitness + concs[None, :] * a222v_remediation
   153|     use_line = (w_post > 0.5) & np.isfinite(w_fitness)
   154|     sm = np.where(use_line[:, None],
   155|                   w_fitness[:, None] + concs[None, :] * w_remediation[:, None],
   156|                   w_mean_score[:, None])
   157|     expected = sm * a222v_line
   158|     if correction is not None:
   159|         cb, cr = correction
   160|         expected = expected + cb(w_fitness)[:, None] + cr(w_fitness)[:, None] * concs[None, :]
   161| 
   162|     valid = np.isfinite(m_score) & np.isfinite(m_se) & (m_se > 0) & np.isfinite(expected)
   163|     resid = np.where(valid, m_score - expected, np.nan)
   164| 
   165|     e_b, e_r, df = wls_line(resid, m_se, concs, valid)
   166| 
   167|     safe_se = np.where(valid, m_se, 1.0)
   168|     obs = np.where(valid, resid, 0.0)
   169|     logl = _gauss_logl(e_b[:, None] + concs[None, :] * e_r[:, None], obs, safe_se, valid)

   171|     # static model: constant offset only (weighted mean residual), r fixed at 0

   173|     with np.errstate(invalid="ignore", divide="ignore"):
